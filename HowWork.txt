    window = WindowBuilder::new()
        .with_title("game") Имя программы

        .build(&event_loop) и указываем наш event_loop
        .unwrap();

// Создаём instance (экзепляр)
    //InstanceDescriptor::default() - значит мы выбираем стандартные настройки
    instance = wgpu::Instance::new(InstanceDescriptor::default());

// Создаём surface (поверхность окна)
    surface = instance.create_surface(&window) указываем наше окно
        .expect("Failed to create surface"); в случае ошибки выводим в консоль

//addaptor(PhysicalDevice)
    //задаём настройки видеокарты
    addapter_option = wgpu::RequestAdapterOptions { 
        //выбирает адаптер, совместимый с этой поверхностью (обычно дискретная видеокарта)
        compatible_surface : Some(&surface),
        //всё остальное поумолчанию
        ..Default::default()
    };

    //запрашиваем физ устройство
    addapter_future = instance.request_adapter(&addapter_option);
    //ожидаем
    addapter = pollster::block_on(addapter_future).unwrap(); //ожидаем через ansync main
    //выводим информацию
    println!("{}",addapter.get_info().name);

//device(LogicalDevice)
    (device, queue) = задаём устройство + очередь (queue.submit(commands);)

            required_limits: Limits::default(),

            label: None, Имя для отладки

        None,
