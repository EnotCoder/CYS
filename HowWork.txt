struct DepthBuffer {
    _texture: wgpu::Texture, // Сама текстура глубины
    view: wgpu::TextureView, // Представление текстуры (как смотреть)
}

impl DepthBuffer {
    new() <- (device: &wgpu::Device, size: winit::dpi::PhysicalSize<u32>) -> DepthBuffer (
        texture = device.create_texture(&wgpu::TextureDescriptor {
            label: Some("Depth Texture"), Отладка
            size: wgpu::Extent3d { Размеры текстуры (ширина, высота, глубина)
                width: size.width,
                height: size.height,
                depth_or_array_layers: 1,
            },
            mip_level_count: 1, Без mip-уровней (не нужны для depth)
            sample_count: 1, Без мультисэмплинга
            dimension: wgpu::TextureDimension::D2, 2D текстура
            //Каждый пиксель хранит 32-битное число с плавающей точкой (от 0.0 до 1.0)
            format: wgpu::TextureFormat::Depth32Float, 32-битный float для глубины
            //Можно использовать как attachment в render pass
            usage: wgpu::TextureUsages::RENDER_ATTACHMENT,
            view_formats: &[], 	Дополнительные форматы для view
        });
        
        //Позволяет читать/писать часть текстуры
        let view = texture.create_view(&wgpu::TextureViewDescriptor::default());
        
        return Self { _texture: texture, view }
    )
    //Когда пользователь меняет размер окна, depth buffer тоже должен измениться
    resize() <- (&mut self, device: &wgpu::Device, new_size: winit::dpi::PhysicalSize<u32>) (
        *self = Self::new(device, new_size); Задаём новый размер
    )
}

create_perspective_matrix() <- (aspect: f32, fov: f32, near: f32, far: f32) -> [f32; 16] (
    let f = 1.0 / (fov * 0.5).tan();
    [
        f / aspect, 0.0, 0.0, 0.0,
        0.0, f, 0.0, 0.0,
        0.0, 0.0, far / (far - near), 1.0,
        0.0, 0.0, -far * near / (far - near), 0.0,
    ]
)

struct Buffers{
    projection,
    uniform_buffer,
    depth_buffer,
    depth_stencil,
    bind_group_layout,
    bind_groupprojection,
}

init_buffers() <- (
    window_size, <- window.inner_size()
    translation, <- [-0.2, 0.0, 0.0, 0.0]
    rotation, <- [-0.2, 0.0, 0.0, 0.0]
    &device, <- LogicalDevice
) -> Buffers(

    //800 на 800 = Соотношение сторон окна 1.0
    aspect = window_size.width as f32 / window_size.height as f32;

    // aspect, π/4 (45°) угол обзора камеры (FOV), 0.1 (ничего ближе 0.1 не видно), 
    //  100.0 (ничего дальше 100 не видно)
    projection = create_perspective_matrix(aspect, std::f32::consts::PI / 4.0, 0.1, 100.0);

    uniforms = Uniforms { вес информации 112 байт
        translation,    <- Позиция камеры / 16 байт
        rotation,       <- Вращение камеры / 16 байт
        projection,     <- Матрица проекции / 64 байт
        use_texture: 1, <- Использовать текстуру (1 = да) / 4 байт
        _padding: [0.0; 3], <- Выравнивание / 12 байт
    };

    // Создаём uniform buffer
    // Выделяет память
    uniform_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
        label: Some("Uniform Buffer"), для отладки
        //bytemuck::cast_slice -> превращает Rust-структуру в сырые байты
        contents: bytemuck::cast_slice(&[uniforms]), загрузка данных
        //UNIFORM (для шейдеров) и COPY_DST (можно обновлять)
        usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
    });


    // Создаём bind group layout (описывает доступ к uniform буферу в шейдере)
    bind_group_layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
        label: Some("Bind Group Layout"), Для отладки
        entries: &[ 
            wgpu::BindGroupLayoutEntry {
                binding: 0, Номер слота, в шейдере будет @binding(0)
                //Доступен как в VERTEX шейдере так и в FRAGMENT шейдере
                visibility: wgpu::ShaderStages::VERTEX | wgpu::ShaderStages::FRAGMENT,
                ty: wgpu::BindingType::Buffer { Указывает тип данных и их свойства
                    //Uniform (Матрицы, параметры камеры), Storage(Большие массивы данных),
                    //  Dynamic(Индивидуальные данные для каждого объекта)
                    ty: wgpu::BufferBindingType::Uniform,
                    has_dynamic_offset: false, //Можно ли менять смещение в буфере
                    min_binding_size: None, // Минимальный размер буфера -> автоматически
                },
                count: None, Для массивов текстур или буферов
            },
        ],
    });

    // Создаём bind group (связывает uniform буфер с шейдером)
    bind_groupprojection = device.create_bind_group(&wgpu::BindGroupDescriptor {
        label: Some("Bind Group"), Отладка
        layout: &bind_group_layout, Передаём layout
        entries: &[
            wgpu::BindGroupEntry {
                binding: 0, // Номер слота (совпадает с layout)
                //as_entire_binding() - говорит, что используем весь буфер целиком
                resource: uniform_buffer.as_entire_binding(), // Что передаём 
            },
        ],
    });

    // Создаём depth texture - текстура, которая хранит для каждого пикселя его глубину
    depth_buffer = DepthBuffer::new(&device, window_size);

    // Настройка depth_stencil для render pipeline
    depth_stencil = wgpu::DepthStencilState {
        format: wgpu::TextureFormat::Depth32Float, 32-битный float для глубины
        depth_write_enabled: true, Разрешить запись глубины 
        depth_compare: wgpu::CompareFunction::Less, Сравнение: new_depth < old_depth
        stencil: wgpu::StencilState::default(), Без трафаретного буфера
        bias: wgpu::DepthBiasState::default(), Без смещения глубины
    };

    return Buffers {
        projection,
        uniform_buffer,
        depth_buffer,
        depth_stencil,
        bind_group_layout,
        bind_groupprojection,
    }
)

struct Model_obj {
    vertices: Vec<Vertex>,
    indices: Vec<u32>,
}

load_obj_simple() <- (path: &str) -> Result<Model_obj, String> {

    file = File::open(path).map_err(|e| format!("Failed to open file: {}", e))?;
    reader = BufReader::new(file);
    
    positions = Vec::new();
    tex_coords = Vec::new();
    face_indices = Vec::new();
    
    for line in reader.lines() {
        line = line.map_err(|e| format!("Failed to read line: {}", e))?;
        parts: Vec<&str> = line.split_whitespace().collect();
        
        if parts.is_empty() {
            continue;
        }
        
        match parts[0] {
            "v" => {  // Вершина (позиция)
                x = parts[1].parse::<f32>().unwrap_or(0.0);
                y = parts[2].parse::<f32>().unwrap_or(0.0);
                z = parts[3].parse::<f32>().unwrap_or(0.0);
                positions.push([x, y, z]);
            }
            "vt" => {  // Материал/цвет
                u = parts[1].parse::<f32>().unwrap_or(0.0);
                v = parts[2].parse::<f32>().unwrap_or(0.0);

                tex_coords.push([u, v]);
            }
            "f" => {  // Грань
                for i in 1..parts.len() {
                    face_part = parts[i];
                    // indices = ["5", "3"]
                    indices: Vec<&str> = face_part.split('/').collect();
                    
                    // indices[0] = "5"
                    // parse::<usize>() = 5
                    // 5 - 1 = 4
                    // pos_idx = 4 (индекс вершины в массиве positions)
                    pos_idx = indices[0].parse::<usize>().unwrap_or(0) - 1;
                    
                    // Индекс текстурных координат
                    tex_idx = if indices.len() > 1 && !indices[1].is_empty() {
                        Some(indices[1].parse::<usize>().unwrap_or(0) - 1)
                    } else {
                        None
                    };
                    
                    face_indices.push((pos_idx, tex_idx));
                }
            }
            _ => {}
        }
    }
    
    // Создаём вершины с правильными цветами
    vertices = Vec::new();
    for (pos_idx, tex_idx_opt) in face_indices {
        pos = positions[pos_idx];
        
        // Получаем UV координаты, если они есть
        tex = match tex_idx_opt {
            Some(tex_idx) if tex_idx < tex_coords.len() => tex_coords[tex_idx],
            _ => [0.0, 0.0],  // Значение по умолчанию, если UV нет
        };
        
        vertices.push(Vertex {
            position: [pos[0], pos[1], pos[2]],
            tex_coord: tex,
        });
    }
    
    indices: Vec<u32> = (0..vertices.len() as u32).collect();
    

    return Ok(Model_obj { vertices, indices })
}

struct ModelInstance {
    vertices: Vec<Vertex>,
    indices: Vec<u16>,
    translation: [f32; 4],
    translation_base: [f32; 4],
    rotation: [f32; 4],
    vertex_buffer: wgpu::Buffer,
    index_buffer: wgpu::Buffer,
    index_count: u32,
    uniform_buffer: wgpu::Buffer,
    bind_group: wgpu::BindGroup,
    texture_bind_group: wgpu::BindGroup,
}

impl ModelInstance {
    new() <- (
        path: &str,
        device: &wgpu::Device,
        queue: &wgpu::Queue,
        translation: [f32; 4],
        translation_base: [f32; 4],
        rotation: [f32; 4],
        projection: [f32; 16],
        texture_path: &str,
    )-> ModelInstance {
        model_result = load_obj_simple(path);

        (vertices, indices_u32) = match model_result {
            Ok(model) => {
                println!("Loaded model: {} vertices, {} indices", 
                        model.vertices.len(), model.indices.len());
                
                (model.vertices, model.indices)
            }
            Err(e) => {
                eprintln!("Failed to load model: {}", e);
                
                default_vertices = vec![
                    Vertex { position: [-0.5, 0.5, 0.0], tex_coord: [0.0, 0.0] },
                    Vertex { position: [-0.5, -0.5, 0.0], tex_coord: [0.0, 1.0] },
                    Vertex { position: [0.5, -0.5, 0.0], tex_coord: [1.0, 1.0] },

                    Vertex { position: [0.5, -0.5, 0.0], tex_coord: [1.0, 1.0] },
                    Vertex { position: [0.5, 0.5, 0.0], tex_coord: [1.0, 0.0] },
                    Vertex { position: [-0.5, 0.5, 0.0], tex_coord: [0.0, 0.0] },
                ];

                default_indices: Vec<u32> = vec![0, 1, 2, 3, 4, 5];

                (default_vertices, default_indices)
            }
        };

        indices: Vec<u16> = indices_u32.iter().map(|&i| i as u16).collect();
        index_count = indices.len() as u32;

        // Создаём буфер вершин
        vertex_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
            label: Some(&format!("Vertex Buffer")),
            contents: bytemuck::cast_slice(&vertices),
            usage: wgpu::BufferUsages::VERTEX,
        });
        
        // Создаём буфер индексов
        index_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
            label: Some(&format!("Index Buffer")),
            contents: bytemuck::cast_slice(&indices),
            usage: wgpu::BufferUsages::INDEX,
        });

        // Создаём свой uniform buffer для этой модели
        uniforms = Uniforms { translation, rotation, projection, use_texture: 1, _padding: [0.0; 3]};
        uniform_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
            label: Some("Model Uniform Buffer"),
            contents: bytemuck::cast_slice(&[uniforms]),
            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
        });
        
        // Создаём bind group layout
        bind_group_layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
            label: Some("Model Bind Group Layout"),
            entries: &[wgpu::BindGroupLayoutEntry {
                binding: 0,
                visibility: wgpu::ShaderStages::VERTEX | wgpu::ShaderStages::FRAGMENT,
                ty: wgpu::BindingType::Buffer {
                    ty: wgpu::BufferBindingType::Uniform,
                    has_dynamic_offset: false,
                    min_binding_size: None,
                },
                count: None,
            }],
        });
        
        // Создаём bind group
        bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
            label: Some("Model Bind Group"),
            layout: &bind_group_layout,
            entries: &[wgpu::BindGroupEntry {
                binding: 0,
                resource: uniform_buffer.as_entire_binding(),
            }],
        });

        //texture

        texture = match Texture::from_path(device, queue, texture_path, "model_texture") {
            Ok(t) => t,
            Err(e) => {
                
                match Texture::from_path(device, queue, "tex/null.png", "null_texture") {
                    Ok(t) => {
                        t
                    }
                    Err(e) => {
                        panic!("Cannot continue without texture");
                    }
                }
            }
        };

        t_bind_group_layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
            label: Some("Texture Bind Group Layout"),
            entries: &[wgpu::BindGroupLayoutEntry {
                binding: 0,
                visibility: wgpu::ShaderStages::FRAGMENT,
                ty: wgpu::BindingType::Texture {
                    multisampled: false,
                    view_dimension: wgpu::TextureViewDimension::D2,
                    sample_type: wgpu::TextureSampleType::Float { filterable: true },
                },
                count: None,
                },
                wgpu::BindGroupLayoutEntry {
                    binding: 1,
                    visibility: wgpu::ShaderStages::FRAGMENT,
                    ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering),
                    count: None,
                },
            ],
        });
        

        t_bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
            label: Some("Texture Bind Group"),
            layout: &t_bind_group_layout,
            entries: &[                
                wgpu::BindGroupEntry {
                    binding: 0,
                    resource: wgpu::BindingResource::TextureView(&texture.view),
                },
                wgpu::BindGroupEntry {
                    binding: 1,
                    resource: wgpu::BindingResource::Sampler(&texture.sampler),
                },
            ],
        });
        
        Self{
            vertices,
            indices,
            translation,
            translation_base,
            rotation,
            vertex_buffer,
            index_buffer,
            index_count,
            uniform_buffer,
            bind_group,
            texture_bind_group: t_bind_group,
        }
    }

    update_transform() <- (&self, queue: &wgpu::Queue, projection: [f32; 16], use_texture:i32) {
        uniforms = Uniforms { 
            translation: self.translation, 
            rotation: self.rotation, 
            projection,
            use_texture,
            _padding: [0.0; 3],
        };
        queue.write_buffer(&self.uniform_buffer, 0, bytemuck::cast_slice(&[uniforms]));
    }
}

struct Texture {
    texture: wgpu::Texture,      // Сама текстура на GPU
    view: wgpu::TextureView,     // Представление текстуры
    sampler: wgpu::Sampler,      // Сэмплер (как читать текстуру)
}

impl Texture {
    from_bytes () <- (
        device: &wgpu::Device,
        queue: &wgpu::Queue,
        bytes: &[u8],
        label: &str,
    ) -> Result<Self, Box<dyn std::error::Error>> {
        // 1. Загружаем изображение из байтов с помощью библиотеки image
        img = image::load_from_memory(bytes)?;
        // 2. Конвертируем в формат RGBA8 (32 бита на пиксель: R,G,B,A)
        rgba = img.to_rgba8();
        // 3. Получаем размеры изображения
        dimensions = img.dimensions();
        
        // 4. Создаём размер текстуры
        size = wgpu::Extent3d {
            width: dimensions.0,
            height: dimensions.1,
            depth_or_array_layers: 1,  // 2D текстура, не массив
        };
        
        // 5. Создаём текстуру на GPU
        texture = device.create_texture(&wgpu::TextureDescriptor {
            label: Some(label),
            size,
            mip_level_count: 1,              // Без MIP-уровней (пока)
            sample_count: 1,                 // Без мультисэмплинга
            dimension: wgpu::TextureDimension::D2,  // 2D текстура
            format: wgpu::TextureFormat::Rgba8UnormSrgb,  // Формат RGBA8
            usage: wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
            view_formats: &[],
        });
        
        // 6. Записываем данные пикселей в текстуру
        queue.write_texture(
            wgpu::ImageCopyTexture {
                texture: &texture,
                mip_level: 0,
                origin: wgpu::Origin3d::ZERO,
                aspect: wgpu::TextureAspect::All,
            },
            &rgba,  // Данные пикселей
            wgpu::ImageDataLayout {
                offset: 0,
                bytes_per_row: Some(4 * dimensions.0),  // 4 байта на пиксель * ширину
                rows_per_image: Some(dimensions.1),
            },
            size,
        );
        
        // 7. Создаём представление текстуры (способ доступа)
        view = texture.create_view(&wgpu::TextureViewDescriptor::default());
        
        // 8. Создаём сэмплер (определяет как текстура накладывается)
        sampler = device.create_sampler(&wgpu::SamplerDescriptor {
            address_mode_u: wgpu::AddressMode::Repeat,   // Повторять по U
            address_mode_v: wgpu::AddressMode::Repeat,   // Повторять по V
            address_mode_w: wgpu::AddressMode::Repeat,   // Повторять по W
            mag_filter: wgpu::FilterMode::Linear,        // Увеличение: линейное
            min_filter: wgpu::FilterMode::Nearest,       // Уменьшение: ближайший
            mipmap_filter: wgpu::FilterMode::Nearest,    // MIP-фильтр
            ..Default::default()
        });
        
        return Ok(Self { texture, view, sampler })
    }
    
    from_path () <- (
        device: &wgpu::Device,
        queue: &wgpu::Queue,
        path: &str,
        label: &str,
    ) -> Result<Self, Box<dyn std::error::Error>> {
        bytes = std::fs::read(path)?;
        return Self::from_bytes(device, queue, &bytes, label)
    }
}
